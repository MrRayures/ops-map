# Outil « Ligne » — Plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ajouter un outil « Ligne » (polyligne ouverte) à OPS Map, avec choix de couleur et de forme du trait (continu / tirets / pointillé), persistée comme les zones.

**Architecture:** Copie fidèle du pattern Zone (data / store / layer / editor + câblage). Chaque outil reste isolé avec son store et sa couche Leaflet. Les différences : polyligne au lieu de polygone, min 2 sommets au lieu de 3, pas de remplissage / masque / play-area / label, nouveau champ `style`.

**Tech Stack:** Astro 6, TypeScript (strict), Leaflet 1.9, Tailwind 4. Aucun runner de test installé — vérification via `npx astro check` (types) + `npm run build`, plus une checklist de test manuel finale.

**Note conventions :** TypeScript strict, pas de `any`, types de retour explicites sur les fonctions exportées, `import type` pour les types. Commits au format `type(scope): description` (voir `.claude/rules/git.md`). Un hook formateur (Prettier) tourne après chaque écriture de fichier — c'est normal.

```ts
import { DEFAULT_LINE_DRAFT, LINE_MIN_VERTICES, type Line, type LineDraft } from '../data/lines';

export type LineMode = 'idle' | 'placing';

export interface LinesState {
  readonly lines: readonly Line[];
  readonly selectedId: string | null;
  readonly mode: LineMode;
  readonly draft: LineDraft;
  readonly draftPoints: readonly [number, number][];
}

type Listener = (state: LinesState) => void;

const listeners = new Set<Listener>();

let state: LinesState = {
  lines: [],
  selectedId: null,
  mode: 'idle',
  draft: { ...DEFAULT_LINE_DRAFT },
  draftPoints: [],
};

function emit(): void {
  for (const listener of listeners) listener(state);
}

function setState(patch: Partial<LinesState>): void {
  state = { ...state, ...patch };
  emit();
}

export function getState(): LinesState {
  return state;
}

export function subscribe(listener: Listener): () => void {
  listeners.add(listener);
  return () => {
    listeners.delete(listener);
  };
}

export function hydrate(lines: readonly Line[]): void {
  setState({
    lines: [...lines],
    selectedId: null,
    mode: 'idle',
    draft: { ...DEFAULT_LINE_DRAFT },
    draftPoints: [],
  });
}

export function setMode(mode: LineMode): void {
  if (state.mode === mode) return;
  if (mode === 'idle') {
    setState({ mode, draftPoints: [] });
  } else {
    setState({ mode, draftPoints: [], selectedId: null });
  }
}

export function updateDraft(patch: Partial<LineDraft>): void {
  setState({ draft: { ...state.draft, ...patch } });
}

export function addDraftPoint(lat: number, lng: number): void {
  if (state.mode !== 'placing') return;
  setState({ draftPoints: [...state.draftPoints, [lat, lng]] });
}

export function finishDraft(): Line | null {
  if (state.mode !== 'placing') return null;
  if (state.draftPoints.length < LINE_MIN_VERTICES) return null;
  const id =
    typeof crypto !== 'undefined' && 'randomUUID' in crypto
      ? crypto.randomUUID()
      : `l-${Date.now()}-${Math.random().toString(36).slice(2, 9)}`;
  const line: Line = {
    id,
    points: state.draftPoints.map((p) => [p[0], p[1]]),
    ...state.draft,
  };
  setState({
    lines: [...state.lines, line],
    selectedId: id,
    mode: 'idle',
    draftPoints: [],
  });
  return line;
}

export function selectLine(id: string | null): void {
  if (state.selectedId === id) return;
  setState({ selectedId: id });
}

export function updateSelected(patch: Partial<LineDraft>): void {
  if (state.selectedId === null) return;
  setState({
    lines: state.lines.map((l) => (l.id === state.selectedId ? { ...l, ...patch } : l)),
  });
}

export function moveSelectedVertex(index: number, lat: number, lng: number): void {
  if (state.selectedId === null) return;
  setState({
    lines: state.lines.map((l) => {
      if (l.id !== state.selectedId) return l;
      const points = l.points.map((p, i) => (i === index ? [lat, lng] : p)) as [number, number][];
      return { ...l, points };
    }),
  });
}

export function insertVertexAfter(index: number, lat: number, lng: number): void {
  if (state.selectedId === null) return;
  setState({
    lines: state.lines.map((l) => {
      if (l.id !== state.selectedId) return l;
      const points: [number, number][] = [...l.points];
      const insertAt = Math.min(Math.max(index + 1, 0), points.length);
      points.splice(insertAt, 0, [lat, lng]);
      return { ...l, points };
    }),
  });
}

export function removeVertex(index: number): void {
  if (state.selectedId === null) return;
  setState({
    lines: state.lines.map((l) => {
      if (l.id !== state.selectedId) return l;
      // Keep the polyline valid (>= 2 vertices).
      if (l.points.length <= LINE_MIN_VERTICES) return l;
      const points = l.points.filter((_, i) => i !== index) as [number, number][];
      return { ...l, points };
    }),
  });
}

export function deleteSelected(): void {
  if (state.selectedId === null) return;
  setState({
    lines: state.lines.filter((l) => l.id !== state.selectedId),
    selectedId: null,
  });
}
```

**Files:**

- [ ] **Step 1: Créer le composant**

Calqué sur `ZoneLayer.astro`, simplifié (pas de masque, play-area, label, centroïde) et avec polyligne ouverte.

```astro
---
// LineLayer is a side-effect component: it has no DOM of its own.
// It renders lines (polylines) on the Leaflet map, plus vertex drag
// handles for the selected line and a draft preview during placing mode.
---

<script>
  import L from "leaflet";
  import {
    subscribe,
    getState,
    selectLine,
    moveSelectedVertex,
    insertVertexAfter,
    removeVertex,
  } from "../lib/lines-store";
  import { whenMapReady } from "../lib/leaflet-map-ref";
  import type { Line, LineStyle } from "../data/lines";

  function dashArrayFor(style: LineStyle): string | undefined {
    if (style === "dashed") return "12 8";
    if (style === "dotted") return "2 6";
    return undefined; // solid
  }

  function strokeColorVar(l: Line): string {
    return `var(--color-${l.color})`;
  }

  function buildVertexIcon(): L.DivIcon {
    return L.divIcon({
      html: `<div class="c-line-vertex"></div>`,
      className: "",
      iconSize: [12, 12],
      iconAnchor: [6, 6],
    });
  }

  function buildMidpointIcon(): L.DivIcon {
    return L.divIcon({
      html: `<div class="c-line-midpoint"></div>`,
      className: "",
      iconSize: [10, 10],
      iconAnchor: [5, 5],
    });
  }

  function buildDraftVertexIcon(): L.DivIcon {
    return L.divIcon({
      html: `<div class="c-line-draft-vertex"></div>`,
      className: "",
      iconSize: [8, 8],
      iconAnchor: [4, 4],
    });
  }

  whenMapReady((map) => {
    const linesGroup = L.layerGroup().addTo(map);
    const handlesGroup = L.layerGroup().addTo(map);
    const draftGroup = L.layerGroup().addTo(map);

    function buildLinePolyline(l: Line, isSelected: boolean): L.Polyline {
      const polyline = L.polyline(l.points, {
        className: `c-line is-color-${l.color}`,
        color: strokeColorVar(l),
        weight: isSelected ? 5 : 3,
        opacity: 0.9,
        dashArray: dashArrayFor(l.style),
      });
      polyline.on("click", (e) => {
        L.DomEvent.stopPropagation(e);
        selectLine(l.id);
      });
      return polyline;
    }

    // Track whether a vertex drag is in progress, so the store-driven
    // re-render doesn't tear down the live handle mid-gesture.
    let dragging = false;

    function renderLines(): void {
      if (dragging) return;
      const state = getState();

      linesGroup.clearLayers();
      handlesGroup.clearLayers();

      for (const l of state.lines) {
        const isSelected = state.selectedId === l.id;

        const polyline = buildLinePolyline(l, isSelected);
        linesGroup.addLayer(polyline);

        if (isSelected) {
          // Mutable vertex array used during drag to update the polyline
          // without involving the store (which would trigger a destructive
          // re-render and break the gesture).
          const livePoints: [number, number][] = l.points.map((p) => [p[0], p[1]]);
          const canRemove = l.points.length > 2;

          l.points.forEach((p, idx) => {
            const handle = L.marker(p, {
              icon: buildVertexIcon(),
              draggable: true,
              keyboard: false,
            });
            handle.on("dragstart", () => {
              dragging = true;
            });
            handle.on("drag", (e) => {
              const ll = (e.target as L.Marker).getLatLng();
              livePoints[idx] = [ll.lat, ll.lng];
              polyline.setLatLngs(livePoints);
            });
            handle.on("dragend", (e) => {
              const ll = (e.target as L.Marker).getLatLng();
              dragging = false;
              moveSelectedVertex(idx, ll.lat, ll.lng);
            });
            handle.on("click", (e) => L.DomEvent.stopPropagation(e));
            handle.on("contextmenu", (e) => {
              L.DomEvent.stopPropagation(e);
              if (e.originalEvent) e.originalEvent.preventDefault();
              if (canRemove) removeVertex(idx);
            });
            handlesGroup.addLayer(handle);
          });

          // Midpoint "insert vertex" handles between consecutive vertices.
          // An open line has no last->first segment, so stop at length - 1.
          for (let i = 0; i < l.points.length - 1; i++) {
            const a = l.points[i];
            const b = l.points[i + 1];
            const mid: [number, number] = [(a[0] + b[0]) / 2, (a[1] + b[1]) / 2];
            const midHandle = L.marker(mid, {
              icon: buildMidpointIcon(),
              keyboard: false,
              interactive: true,
            });
            midHandle.on("click", (e) => {
              L.DomEvent.stopPropagation(e);
              insertVertexAfter(i, mid[0], mid[1]);
            });
            handlesGroup.addLayer(midHandle);
          }
        }
      }
    }

    function renderDraft(): void {
      const state = getState();
      draftGroup.clearLayers();
      if (state.mode !== "placing") return;
      const pts = state.draftPoints;
      if (pts.length === 0) return;

      if (pts.length >= 2) {
        draftGroup.addLayer(
          L.polyline(pts as [number, number][], {
            color: `var(--color-${state.draft.color})`,
            weight: 3,
            opacity: 0.9,
            dashArray: "4 4",
            interactive: false,
          }),
        );
      }

      // Small dots at each placed vertex.
      for (const p of pts) {
        draftGroup.addLayer(
          L.marker(p, { icon: buildDraftVertexIcon(), interactive: false }),
        );
      }
    }

    function syncAll(): void {
      renderLines();
      renderDraft();
    }

    syncAll();
    subscribe(syncAll);
  });
</script>
```

    setMode,
    subscribe,
    updateDraft,
    updateSelected,

  import { bindColorSwatches, reflectChecked } from "../lib/editor-binding";

  const root = document.querySelector<HTMLElement>("[data-line-editor]");
  if (root) {
    const idle = root.querySelector<HTMLElement>("[data-line-editor-idle]")!;
    const form = root.querySelector<HTMLElement>("[data-line-editor-form]")!;
    const statusEl = root.querySelector<HTMLElement>("[data-line-form-status]")!;
    const placingHint = root.querySelector<HTMLElement>("[data-line-placing-hint]")!;
    const editHint = root.querySelector<HTMLElement>("[data-line-edit-hint]")!;
    const startBtn = root.querySelector<HTMLButtonElement>(
      'button[data-action="line-start-placing"]',
    )!;
    const cancelBtn = root.querySelector<HTMLButtonElement>(
      'button[data-action="line-cancel-or-deselect"]',
    )!;
    const finishBtn = root.querySelector<HTMLButtonElement>(
      'button[data-action="line-finish"]',
    )!;
    const colorBtns = root.querySelectorAll<HTMLButtonElement>(
      'button[data-line-field="color"]',
    );
    const styleBtns = root.querySelectorAll<HTMLButtonElement>(
      'button[data-line-field="style"]',
    );
    const deleteWrap = root.querySelector<HTMLElement>("[data-line-delete-wrap]")!;
    const deleteBtn = root.querySelector<HTMLButtonElement>(
      'button[data-action="line-delete"]',
    )!;

    const store = { getState, updateSelected, updateDraft };

    startBtn.addEventListener("click", () => setMode("placing"));

    function closeEditor(): void {
      const { mode, selectedId, draftPoints } = getState();
      if (mode === "placing") {
        if (draftPoints.length >= 2) {
          finishDraft();
        } else {
          setMode("idle");
        }
      } else if (selectedId) {
        selectLine(null);
      }
    }

    cancelBtn.addEventListener("click", () => {
      const { mode } = getState();
      if (mode === "placing") setMode("idle");
      else selectLine(null);
    });
    finishBtn.addEventListener("click", closeEditor);

    bindColorSwatches<{ color: MarkerColor }>(colorBtns, store, "color");
    bindColorSwatches<{ style: LineStyle }>(styleBtns, store, "style");

    deleteBtn.addEventListener("click", () => {
      deleteSelected();
    });

    function render(): void {
      const state = getState();
      const showIdle = state.mode === "idle" && state.selectedId === null;
      idle.hidden = !showIdle;
      form.hidden = showIdle;
      deleteWrap.hidden = !state.selectedId;
      placingHint.hidden = state.mode !== "placing";
      editHint.hidden = state.selectedId === null;

      if (showIdle) return;

      const source = state.selectedId
        ? state.lines.find((l) => l.id === state.selectedId)
        : null;
      const data = source ?? state.draft;

      statusEl.textContent =
        state.mode === "placing"
          ? `Placement (${state.draftPoints.length} point${state.draftPoints.length > 1 ? "s" : ""})`
          : "Ligne sélectionnée";

      cancelBtn.setAttribute(
        "aria-label",
        state.mode === "placing" ? "Annuler le placement" : "Désélectionner",
      );

      finishBtn.disabled =
        state.mode === "placing" && state.draftPoints.length < 2;

      reflectChecked(colorBtns, data.color);
      reflectChecked(styleBtns, data.style);
    }

    render();
    subscribe(render);
  }
</script>
